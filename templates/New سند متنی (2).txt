    <script>
        let currentUser = "";

        function registerUser() {
            const val = document.getElementById('username-input').value.trim();
            if (!val) return alert("لطفاً یک نام کاربری وارد کنید!");
            currentUser = val;
            document.getElementById('display-name').innerText = currentUser;
            document.getElementById('user-avatar').innerText = currentUser[0].toUpperCase();
            document.getElementById('login-modal').classList.add('hidden');
            fetchMessages();
        }

        function toggleModal(id) {
            document.getElementById(id).classList.toggle('hidden');
        }

        function addEmoji(emoji) {
            document.getElementById('msg-input').value += emoji;
        }

        async function sendMessage() {
            const input = document.getElementById('msg-input');
            const text = input.value.trim();
            if (!text) return;

            const res = await fetch('/api/send', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({username: currentUser, text: text})
            });
            const data = await res.json();
            if (res.status === 403) {
                alert(data.message);
            } else {
                input.value = "";
                fetchMessages();
            }
        }

        async function fetchMessages() {
            if (!currentUser) return;
            const res = await fetch('/api/messages');
            const msgs = await res.json();
            const box = document.getElementById('chat-box');
            box.innerHTML = "";

            msgs.forEach(m => {
                if (m.username === currentUser && m.is_premium) {
                    document.getElementById('premium-status').innerHTML = "اشتراک ویژه 💠";
                    document.getElementById('premium-status').style.color = "#00d9ff";
                }
                const div = document.createElement('div');
                div.className = "msg-card " + (m.is_premium ? "msg-premium" : "");
                const badge = m.is_premium ? `<span class="star-badge" title="کاربر پریمیوم">✦☑️</span>` : "";
                div.innerHTML = `
                    <div class="msg-header">
                        <span>${m.username} ${badge}</span>
                        <button class="report-btn" onclick="reportMsg(${m.id})">🚨 گزارش</button>
                    </div>
                    <div>${m.text}</div>
                `;
                box.appendChild(div);
            });
            box.scrollTop = box.scrollHeight;
        }

        async function reportMsg(id) {
            await fetch('/api/report', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({msg_id: id})
            });
            alert("گزارش تخلف برای مدیر در تلگرام ارسال شد.");
        }

        async function buyPremium(plan, paymentUrl) {
            await fetch('/api/request_premium', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({username: currentUser, plan: plan})
            });
            window.open(paymentUrl, '_blank');
            alert("درخواست شما برای مدیر ارسال شد. پس از پرداخت، تیک آبی شما توسط مدیر فعال می‌شود.");
            toggleModal('premium-modal');
        }

        async function sendSupport() {
            const txt = document.getElementById('support-text').value.trim();
            if (!txt) return;
            await fetch('/api/support/send', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({username: currentUser, text: txt})
            });
            document.getElementById('support-text').value = "";
            alert("پیام شما به تلگرام مدیر ارسال شد!");
            toggleModal('support-modal');
        }

        setInterval(fetchMessages, 2000);
    </script>
